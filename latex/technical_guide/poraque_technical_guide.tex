% --- Brand palette ---------------------------------------------------------
% Three colours, and each has one job. The hues run 48 -> 77 -> 160 degrees,
% a single sweep from the logo's yellow through yellow-green into deep green.
%
%   poraqueyellow  the logo's own yellow. ACCENT ONLY -- 1.7:1 on white, so
%                  it is used for rules, badges and marks on dark grounds,
%                  never for text on paper.
%   poraquelime    the step between. Tints and secondary rules.
%   poraquegreen   the anchor: 12.2:1 on white, 8.0:1 against the yellow.
%                  Headings, links, anything that has to be read.
%   poraquecover   COVER GROUND ONLY. Named for the restriction: it is the
%                  one colour with a single permitted use, and a name that
%                  says so is harder to misuse than a comment. Darker than
%                  poraquegreen by 7 points of lightness -- a visible step
%                  rather than a shade of the same thing -- and still green
%                  at hue 163 rather than collapsing to black.
\definecolor{poraqueyellow}{RGB}{255,204,0}
\definecolor{poraquelime}{RGB}{163,198,75}
\definecolor{poraquegreen}{RGB}{15,61,46}
\definecolor{poraquecover}{RGB}{6,35,27}
% Retained so existing uses keep working; both now point into the new palette.
